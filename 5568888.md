from datetime import date
from difflib import SequenceMatcher
import re

WEIGHTS = {
    "name": 0.25,
    "category": 0.20,
    "color": 0.15,
    "location": 0.15,
    "date": 0.10,
    "description": 0.15,
}

CATEGORIES = [
    "電子產品", "證件", "文具",
    "衣物", "水壺", "雨傘", "其他"
]

LABELS = {
    "name": "物品名稱相似",
    "category": "物品類別相同",
    "color": "顏色相近",
    "location": "地點相近",
    "date": "拾獲與遺失日期接近",
    "description": "物品描述相似",
}

found_items = []
lost_items = []


def normalize_text(value):
    """驗證文字型別並移除空白。"""
    if not isinstance(value, str):
        return ""
    return re.sub(r"\s+", "", value.strip().casefold())


def text_similarity(a, b):
    """缺失文字回傳 None，不將未知資料視為不相符。"""
    a = normalize_text(a)
    b = normalize_text(b)

    if not a or not b:
        return None
    if a == b:
        return 1.0
    if a in b or b in a:
        return 0.9

    return SequenceMatcher(None, a, b).ratio()


def parse_date(value):
    """嚴格驗證 YYYY-MM-DD。"""
    if not isinstance(value, str):
        raise ValueError("日期必須是 YYYY-MM-DD 格式")

    if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", value):
        raise ValueError("日期格式必須是 YYYY-MM-DD")

    try:
        parsed = date.fromisoformat(value)
    except ValueError as exc:
        raise ValueError("日期不存在") from exc

    if parsed.isoformat() != value:
        raise ValueError("日期格式不正確")

    return parsed


def date_similarity(a, b):
    try:
        d1 = parse_date(a)
        d2 = parse_date(b)
    except ValueError:
        return None

    days = abs((d1 - d2).days)
    return max(0.0, 1.0 - days / 30.0)


def match_items(lost, found):
    """只用雙方都有提供的特徵計分。"""
    features = {
        "name": text_similarity(
            lost.get("name"), found.get("name")
        ),
        "category": (
            float(lost["category"] == found["category"])
            if lost.get("category") and found.get("category")
            else None
        ),
        "color": text_similarity(
            lost.get("color"), found.get("color")
        ),
        "location": text_similarity(
            lost.get("location"), found.get("location")
        ),
        "date": date_similarity(
            lost.get("date"), found.get("date")
        ),
        "description": text_similarity(
            lost.get("description"), found.get("description")
        ),
    }

    available_weight = sum(
        WEIGHTS[key]
        for key, value in features.items()
        if value is not None
    )

    if available_weight == 0:
        score = 0
    else:
        raw_score = sum(
            WEIGHTS[key] * value
            for key, value in features.items()
            if value is not None
        ) / available_weight

        # 資料太少時限制分數，避免缺失資料造成虛高。
        reliable_features = sum(
            value is not None for value in features.values()
        )
        if reliable_features < 3:
            raw_score = min(raw_score, 0.49)

        score = round(raw_score * 100)

    reasons = [
        LABELS[key]
        for key, value in features.items()
        if value is not None and value >= 0.65
    ]

    if not reasons:
        reasons = ["沒有明顯相符特徵，建議人工確認"]

    return score, reasons, features


def input_item(kind):
    print("\n" + "=" * 45)
    print("【登記撿到的物品】" if kind == "found"
          else "【登記遺失的物品】")
    print("=" * 45)

    while True:
        name = input("物品名稱：").strip()
        if name:
            break
        print("名稱不能空白。")

    print("\n物品類別：")
    for i, category in enumerate(CATEGORIES, 1):
        print(f"{i}. {category}")

    while True:
        choice = input("請輸入類別編號（1-7）：").strip()
        if choice.isdigit() and 1 <= int(choice) <= len(CATEGORIES):
            category = CATEGORIES[int(choice) - 1]
            break
        print("輸入錯誤，請輸入 1 到 7。")

    color = input("物品顏色（不知道可直接 Enter）：").strip()

    while True:
        location = input("遺失／拾獲地點：").strip()
        if location:
            break
        print("地點不能空白。")

    while True:
        item_date = input("日期（YYYY-MM-DD）：").strip()
        try:
            parse_date(item_date)
            break
        except ValueError as exc:
            print(f"日期錯誤：{exc}")

    description = input("其他特徵或描述：").strip()

    return {
        "name": name,
        "category": category,
        "color": color,
        "location": location,
        "date": item_date,
        "description": description,
        "status": "open",
    }


def show_matches(lost):
    candidates = [
        item for item in found_items
        if item["status"] == "open"
    ]

    print("\n" + "=" * 45)
    print("智慧媒合結果")
    print("=" * 45)

    if not candidates:
        print("目前沒有開放中的拾獲物可以比對。")
        return

    results = []
    for found in candidates:
        score, reasons, features = match_items(lost, found)
        results.append({
            "item": found,
            "score": score,
            "reasons": reasons,
            "features": features,
        })

    results.sort(key=lambda x: x["score"], reverse=True)

    for rank, result in enumerate(results, 1):
        found = result["item"]
        score = result["score"]

        if score >= 80:
            verdict = "高度符合，建議優先人工確認"
        elif score >= 60:
            verdict = "可能符合，建議進一步確認"
        elif score >= 40:
            verdict = "部分特徵相似"
        else:
            verdict = "相似度偏低"

        print(f"\n第 {rank} 名：{found['name']}")
        print(f"類別：{found['category']}")
        print(f"顏色：{found['color'] or '未提供'}")
        print(f"地點：{found['location']}")
        print(f"日期：{found['date']}")
        print(f"相似度分數：{score}/100")
        print(f"媒合判斷：{verdict}")
        print("推薦理由：")
        for reason in result["reasons"]:
            print(f"  - {reason}")

        print("各項特徵分數：")
        for key, value in result["features"].items():
            display = "未提供" if value is None else f"{value * 100:.1f}%"
            print(f"  {key}: {display}")


def list_items(items, title):
    print(f"\n【{title}】")
    if not items:
        print("目前沒有資料。")
        return

    for i, item in enumerate(items, 1):
        print(
            f"{i}. {item['name']} | {item['category']} | "
            f"{item['color'] or '未提供'} | {item['location']} | "
            f"{item['date']} | 狀態：{item['status']}"
        )


def main():
    print("=" * 50)
    print("CampusMatch V4 校園失物招領智慧媒合系統")
    print("=" * 50)

    while True:
        print("\n請選擇功能：")
        print("1. 登記撿到的物品")
        print("2. 登記遺失的物品並立即媒合")
        print("3. 查看所有拾獲物")
        print("4. 查看所有遺失物")
        print("5. 結束系統")

        choice = input("請輸入選項（1-5）：").strip()

        if choice == "1":
            item = input_item("found")
            found_items.append(item)
            print("\n拾獲物登記成功！")
            print("目前拾獲物數量：", len(found_items))

            for lost in lost_items:
                print(f"\n比對遺失物：{lost['name']}")
                show_matches(lost)

        elif choice == "2":
            item = input_item("lost")
            lost_items.append(item)
            print("\n遺失物登記成功！")
            show_matches(item)

        elif choice == "3":
            list_items(found_items, "所有拾獲物")

        elif choice == "4":
            list_items(lost_items, "所有遺失物")

        elif choice == "5":
            print("\n感謝使用 CampusMatch V4！")
            break

        else:
            print("選項錯誤，請輸入 1 到 5。")


if __name__ == "__main__":
    main()
